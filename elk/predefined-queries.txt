# ELK Stack Predefined Queries for Security and Application Monitoring
# Use these queries in Kibana's Discover or Dashboard visualizations

## SECURITY MONITORING QUERIES

# 1. Failed Login Attempts in Last Hour
GET security-audit-*/_search
{
  "query": {
    "bool": {
      "must": [
        {
          "term": {
            "audit.eventType": "LOGIN_FAILURE"
          }
        },
        {
          "range": {
            "@timestamp": {
              "gte": "now-1h"
            }
          }
        }
      ]
    }
  },
  "sort": [
    {
      "@timestamp": {
        "order": "desc"
      }
    }
  ],
  "aggs": {
    "failed_by_ip": {
      "terms": {
        "field": "source_ip",
        "size": 10
      }
    },
    "failed_by_user": {
      "terms": {
        "field": "audit.userId",
        "size": 10
      }
    }
  }
}

# 10. Database Query Analysis
GET application-logs-*/_search
{
  "query": {
    "bool": {
      "must": [
        {
          "match": {
            "tags": "sql"
          }
        },
        {
          "range": {
            "@timestamp": {
              "gte": "now-1h"
            }
          }
        }
      ]
    }
  },
  "aggs": {
    "sql_operations": {
      "terms": {
        "script": {
          "source": """
            def message = doc['log_message.keyword'].value;
            if (message.contains('SELECT')) return 'SELECT';
            if (message.contains('INSERT')) return 'INSERT';
            if (message.contains('UPDATE')) return 'UPDATE';
            if (message.contains('DELETE')) return 'DELETE';
            return 'OTHER';
          """
        }
      }
    }
  }
}
